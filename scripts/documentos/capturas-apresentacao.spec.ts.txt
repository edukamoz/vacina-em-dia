// TEMPORÁRIO: gera as capturas da apresentação. Não versionar.
import { mkdirSync } from 'node:fs';
import { test as base } from '@playwright/test';
import { instalarApiFalsa, type EstadoDaApi } from './api-falsa';
const SENHA = 'uma frase longa é melhor';
const test = base.extend<{ api: EstadoDaApi }>({ api: async ({ page }, use) => { await use(await instalarApiFalsa(page)); } });
import type { Page } from '@playwright/test';

const SAIDA = 'D:/Projetos/vacina-em-dia/doctos/capturas/apresentacao';
mkdirSync(SAIDA, { recursive: true });
test.use({ deviceScaleFactor: 2 });

async function entrar(page: Page, api: { consentimentoAceito: boolean }) {
  api.consentimentoAceito = true;
  await page.goto('/entrar');
  await page.getByLabel('E-mail').fill('mariana@exemplo.com.br');
  await page.getByLabel('Senha', { exact: true }).fill(SENHA);
  await page.getByRole('button', { name: 'Entrar' }).click();
  await page.getByText('Nenhuma pessoa cadastrada').waitFor();
}

async function adicionar(page: Page, nome: string, nasc: string) {
  await page.getByRole('button', { name: 'Adicionar pessoa' }).first().click();
  await page.getByLabel('Nome ou apelido').fill(nome);
  await page.getByLabel('Data de nascimento').fill(nasc);
  await page.getByRole('button', { name: 'Salvar e ver as vacinas' }).click();
  await page.getByRole('heading', { name: `Doses de ${nome}` }).waitFor();
}

for (const [rotulo, viewport, escala] of [
  ['web', { width: 1440, height: 900 }, 1],
  ['cel', { width: 390, height: 844 }, 2],
] as const) {
  test(`capturas ${rotulo}`, async ({ page, api }) => {
    await page.setViewportSize(viewport);
    await page.emulateMedia({ reducedMotion: 'no-preference' });
    await page.goto('/apresentacao');
    await page.waitForTimeout(2500);
    await page.screenshot({ path: `${SAIDA}/${rotulo}-landing.png` });
    await page.mouse.wheel(0, 900);
    await page.waitForTimeout(1500);
    await page.screenshot({ path: `${SAIDA}/${rotulo}-landing-2.png` });

    await page.goto('/criar-conta');
    await page.waitForTimeout(1200);
    await page.screenshot({ path: `${SAIDA}/${rotulo}-cadastro.png` });
    await page.goto('/entrar');
    await page.waitForTimeout(1200);
    await page.screenshot({ path: `${SAIDA}/${rotulo}-entrar.png` });

    await entrar(page, api);
    await adicionar(page, 'Maria', '20052025');
    await page.waitForTimeout(1200);
    await page.screenshot({ path: `${SAIDA}/${rotulo}-doses.png` });

    await page.getByRole('button', { name: /BCG, dose única/ }).click();
    await page.waitForTimeout(1000);
    await page.screenshot({ path: `${SAIDA}/${rotulo}-dose.png` });
    await page.goBack();

    await page.getByRole('button', { name: 'Abrir o assistente' }).click();
    const janela = page.getByRole('dialog', { name: 'Assistente' });
    await janela.getByLabel('Sua pergunta').fill('Para que serve a BCG?');
    await janela.getByRole('button', { name: 'Enviar' }).click();
    await janela.getByText(/protege contra formas graves da tuberculose/).waitFor();
    await page.waitForTimeout(800);
    await page.screenshot({ path: `${SAIDA}/${rotulo}-assistente.png` });
  });
}
