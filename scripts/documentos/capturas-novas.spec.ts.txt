// Capturas das telas novas para a Documentação do Usuário (B52 a B57). Não roda no CI: é um
// "teste" de captura. Para usar, copie para `apps/mobile/e2e/zz-capturas-novas.spec.ts`, gere o
// build web com `EXPO_PUBLIC_CAPTURA=1` e uma chave Pix FICTÍCIA (ver scripts/documentos/LEIA-ME.md)
// e rode `npx playwright test e2e/zz-capturas-novas.spec.ts` em apps/mobile. As imagens saem em
// `doctos/capturas/`. A API é a falsa dos testes de ponta a ponta; os dados são fictícios.
import { test as base, expect, type Page } from '@playwright/test';
import { instalarApiFalsa, type EstadoDaApi } from './api-falsa';
import { SENHA } from './base';

// Sem travar o relógio: com a data fixa, as animações de entrada do app não andam e a captura sai em branco.
const test = base.extend<{ api: EstadoDaApi }>({
  api: [async ({ page }, use) => use(await instalarApiFalsa(page)), { auto: true }],
});

const SAIDA = process.env.CAPTURAS_SAIDA ?? '../../doctos/capturas/';
// Foto de perfil fictícia (avatar desenhado), JPEG 256x256.
const FOTO = '__FOTO__';

async function entrarComMaria(page: Page, api: EstadoDaApi) {
  await page.goto('/entrar');
  await page.getByLabel('E-mail').fill('mariana@exemplo.com.br');
  await page.getByLabel('Senha', { exact: true }).fill(SENHA);
  await page.getByRole('button', { name: 'Entrar' }).click();
  await page.getByRole('checkbox', { name: /Li e aceito/ }).click();
  await page.getByRole('checkbox', { name: /responsável legal/ }).click();
  await page.getByRole('button', { name: 'Aceitar e continuar' }).click();
  await expect(page.getByText('Nenhuma pessoa cadastrada')).toBeVisible();
  await page.getByRole('button', { name: 'Adicionar pessoa' }).click();
  await page.getByLabel('Nome ou apelido').fill('Maria');
  await page.getByLabel('Data de nascimento').fill('10022026');
  await page.getByRole('button', { name: 'Salvar e ver as vacinas' }).click();
  await expect(page.getByRole('heading', { name: 'Doses de Maria' })).toBeVisible();
  expect(api.consentimentoAceito).toBe(true);
}

async function foto(page: Page, arquivo: string) {
  await page.waitForTimeout(1200);
  await page.screenshot({ path: `${SAIDA}${arquivo}.png` });
}

test.describe('Capturas do celular (390 x 844)', () => {
  test.use({ viewport: { width: 390, height: 844 }, isMobile: true, hasTouch: true, deviceScaleFactor: 2 });

  test('telas novas no celular', async ({ page, api }) => {
    await entrarComMaria(page, api);

    // Janela Adicionar dose.
    await page.getByRole('button', { name: 'Adicionar dose' }).first().click();
    await expect(page.getByRole('dialog')).toBeVisible();
    await foto(page, 'celular-13-janela-dose');
    await page.getByRole('button', { name: 'Fechar', exact: true }).click();

    // Ler caderneta por foto (leitura simulada com dados fictícios).
    await page.getByRole('button', { name: 'Ler caderneta por foto' }).click();
    await page.getByRole('button', { name: 'Tirar foto' }).click();
    await expect(page.getByText('Doses encontradas')).toBeVisible();
    await foto(page, 'celular-14-caderneta');
    await page.getByRole('button', { name: 'Fechar', exact: true }).click();

    // Janela Adicionar pessoa.
    await page.getByRole('link', { name: 'Família' }).click();
    await page.getByRole('button', { name: 'Adicionar pessoa' }).click();
    await expect(page.getByRole('dialog')).toBeVisible();
    await foto(page, 'celular-12-janela-pessoa');
    await page.getByRole('button', { name: 'Fechar', exact: true }).click();

    // Conta com foto de perfil e Pix.
    api.foto = { contentType: 'image/jpeg', data: FOTO };
    await page.getByRole('link', { name: 'Conta' }).click();
    await expect(page.getByRole('button', { name: 'Trocar foto' })).toBeVisible();
    await foto(page, 'celular-10-conta-foto');
    await page.getByRole('button', { name: 'Doar com Pix' }).scrollIntoViewIfNeeded();
    await page.getByRole('button', { name: 'Doar com Pix' }).click();
    await page.getByRole('button', { name: 'R$ 10' }).click();
    await expect(page.getByLabel('QR Code do Pix')).toBeVisible();
    await foto(page, 'celular-11-pix');
  });
});

test.describe('Capturas da apresentação na web', () => {
  test('site grande (1280 x 800)', async ({ page }) => {
    await page.setViewportSize({ width: 1280, height: 800 });
    await page.goto('/apresentacao');
    await expect(page.getByRole('button', { name: 'Criar conta' }).first()).toBeVisible();
    await page.waitForTimeout(2500);
    await page.screenshot({ path: `${SAIDA}web-10-apresentacao-grande.png` });
  });

  test('site pequeno (390 x 844)', async ({ page }) => {
    await page.setViewportSize({ width: 390, height: 844 });
    await page.goto('/apresentacao');
    await expect(page.getByRole('button', { name: 'Criar conta' }).first()).toBeVisible();
    await page.waitForTimeout(2500);
    await page.screenshot({ path: `${SAIDA}web-11-apresentacao-pequena.png` });
  });
});
